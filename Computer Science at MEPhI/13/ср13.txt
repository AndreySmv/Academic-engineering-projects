int load_int(int **din_mas)
{
    int k;
    printf("\n\nenter array size: k = ");
    scanf("%d", &k);
    *din_mas = malloc(k * sizeof(int));
    if (*din_mas == NULL)
    {
        printf("\n\nmemory allocation error\n.");
        exit(EXIT_FAILURE);
    }
    printf("\ninput array: ");
    for (int i = 0; i < k; i++)
    {
        scanf(" %d", &(*din_mas)[i]);
    }
    printf("\n");
    return k;
}
void print_mas(int **din_mas, int zzz)
{
    printf("\n\narray: ");
    for (int i = 0; i < zzz; i++)
    {
        printf("%d ", (*din_mas)[i]);
    }
    printf("\n");
}
void transposition(int **din_mas, int k)
{
    int n;
    printf("\n\nenter n for transposition: n = ");
    scanf("%d", &n);
    if(2*n>=k)
    {
        if(k%2==0)
        {
            printf("\n2n>=k. transposition k elements\n");
            n=(int)(0.5*(float)k);
        }
        if(k%2!=0)
        {
            printf("\n2n>=k. transposition k-1 elements\n");
            n=(int)(0.5*(float)(k-1));
        }
    }
    int mas_1[n], mas_2[n], count_1=-1, count_2=-1, count=-1;
    for(int i=0; i<2*n; i++)
    {
        if((i+1)<=n)
        {
            count_1++;
            mas_1[count_1]=(*din_mas)[i];
        }
        else
        {
            count_2++;
            mas_2[count_2]=(*din_mas)[i];
        }
    }
    count_1=-1; count_2=-1;
    for(int i=0; i<2*n; i++)
    {
        if(i%2==0)
        {
            count_1++;
            (*din_mas)[i]=mas_1[count_1];
        }
        else
        {
            count_2++;
            (*din_mas)[i]=mas_2[count_2];
        }
    }
    printf("\n");
}
void save_to_file(int **din_mas, int k)
{
    FILE *file=fopen("din_mas.txt", "w");
    printf("\n\nwrite to file\n");
    for(int i=0; i<k; i++)
	{
	    if(i==0)
        {
            fprintf(file, "%d", (*din_mas)[i]);
        }
        else
        {
            fprintf(file, " %d", (*din_mas)[i]);
        }
	}
	fclose(file);
}
void read_from_file(int k)
{
    FILE *file=fopen("din_mas.txt", "r");
    printf("\n\nread from file\n");
    printf("array: ");
    char ch;
    while((ch=getc(file))!=EOF)
    {
        putchar(ch);
    }
    fclose(file);
}
int delete_from_array(int **din_mas, int **q, int k)
{
    printf("\n\nlocate remove element = ");
    int j, loc, count_del=-1;
    scanf("%d", &j);
    while(j<=k && count_del<k-1)
    {
        count_del++;
        loc=j+count_del;
        if(loc==1)
        {
            for(int i=0; i<k-count_del; i++)
            {
                (*din_mas)[i]=(*din_mas)[i+1];
            }
        }
        else if(loc>1 && loc<k-count_del)
        {
            for(int i=0; i<k-count_del-loc; i++)
            {
                (*din_mas)[loc-1+i]=(*din_mas)[loc+i];
            }
        }
        if(loc>=1 && j<=k)
        {
            *q=(int*) (int*) realloc(*din_mas, (k-count_del-1)*sizeof(int));
        }
        if (*q!=NULL)
            {
                *din_mas=*q;
            }
        printf("\n\nlocate remove element = ");
        scanf("%d", &j);
    }
    printf("\n");
    return k-count_del-1;
}
int append_from_array(int **din_mas, int **q, int k)
{
    int j, loc, indx, rem, mem, dim, f, count_app, count_elements;
    printf("\n\nenter the quantity of elements: ");
    scanf("%d", &count_elements);
    printf("\nlocate append elements: ");
    scanf("%d", &j);
    //mem=j;
    for(int i=1; i<=count_elements; i++)
    {
        count_app=i-1;
        *q=(int*) (int*) realloc(*din_mas, (k+count_app+1)*sizeof(int));
        if (*q==NULL)
            {
                printf("\n\nmemory allocation error.");
                exit(EXIT_FAILURE);
            }
        printf("\n%d element for append: ", i);
        scanf("%d", &f);
        *din_mas=*q;
        count_app++;
        dim=k+count_app;
        rem=(k+(count_app-1))-j+1;
        for(int i=0; i<rem; i++)
        {
            (*din_mas)[dim-1-i]=(*din_mas)[dim-2-i];
        }
        (*din_mas)[j-1]=f;
    }
    printf("\n");
    return k+count_elements;
}
int main()
{
    int k, n, d, *din_mas, *q, *qq;
