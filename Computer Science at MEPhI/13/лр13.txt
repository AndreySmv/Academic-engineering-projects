#include <stdio.h>
#include <stdlib.h>
void del(int *p, int *q, int n, int k)
{
    if(k==1) //начало
    {
        for(int i=0; i<n-1; i++)
        {
            p[i]=p[i+1];
        }
    }
    else if(k>1 && k<n) //внутри
    {
        for(int i=0; i<n-k; i++)
        {
            p[k-1+i]=p[k+i];
        }
    }
    if(k>=1 && k<=n)
    q=(int*) (int*) realloc(p, (n-1)*sizeof(int));
    if (q!=NULL) {p=q;}
}
void transposition(int *p, int n, int k, int i, int j)
{
    if(i!=j)
    {
        int t;
        t=p[i-1];
        p[i-1]=p[j-1];
        p[j-1]=t;
    }
}
int main() 
{
    int n, k, i, j, *p, *q;
    scanf("%d", &n);
    p=(int*) malloc(n*sizeof(int)); 
    //if(pa==NULL) {exit(EXIT_FAILURE);}
    for(int i=0; i<n; i++) {scanf("%d", p+i);}
    scanf("%d", &k); scanf("%d %d", &i, &j);
    for(int i=0; i<n; i++) {printf("%3d", i+1);}
    printf("\n");
    for(int i=0; i<n; i++) {printf("%3d", *(p+i));}
    del(p, q, n, k);
    printf("\n\n");
    transposition(p, n, k, i, j);
    for(int i=0; i<n-1; i++) {printf("%3d", i+1);}
    printf("\n");
    for(int i=0; i<n-1; i++) {printf("%3d", *(p+i));}
    printf("\n");
    return 0;
}
