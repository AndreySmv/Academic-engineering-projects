#include <stdio.h>

void fill_matrix(int m, int n, int mtrx[m][n]) 
{
    int value;
    for (int j = 0; j < n; j++)
    {
        if (j % 2 == 0) // Если номер столбца четный, начинаем с нижнего элемента
        {
            for (int i = m - 1; i >= 0; i--)
            {
                scanf("%d", &value);
                mtrx[i][n - 1 - j] = value;
            }
        }
        else // Если номер столбца нечетный, начинаем с верхнего элемента
        {
            for (int i = 0; i < m; i++)
            {
                scanf("%d", &value);
                mtrx[i][n - 1 - j] = value;
            }
        }
    }
}

void find_min_values(int m, int n, int mtrx[m][n], int ar_min[m], int *min) 
{
    for (int i=0; i<m; i++) 
    {
        ar_min[i] = mtrx[i][0];
        for (int j = 0; j < n; j++)
            if (ar_min[i] > mtrx[i][j]) ar_min[i] = mtrx[i][j];
    }

    *min = ar_min[0];
    for (int i = 0; i < m; i++)
        if (*min > ar_min[i]) *min = ar_min[i];
}

void find_min_values_2(int m, int n, int* mtrx, int* ar_min, int *min) {
    for (int i = 0; i < m; i++) {
        *(ar_min + i) = *(mtrx + i * n);
        for (int j = 0; j < n; j++)
            if (*(ar_min + i) > *(mtrx + i * n + j)) *(ar_min + i) = *(mtrx + i * n + j);
    }

    *min = *ar_min;
    for (int i = 0; i < m; i++)
        if (*min > *(ar_min + i)) *min = *(ar_min + i);
}

void print_matrix(int m, int n, int mtrx[m][n]) 
{
    for (int i=0; i<m; i++) 
    {
        for (int j=0; j<n; j++)
            printf("%3d", mtrx[i][j]);
        printf("\n");
    }
}
