#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void print_matrix(int **matrix, int rows, int cols)
{
    for (int i=0; i<rows; i++)
    {
        for (int j=0; j<cols; j++)
        {
            printf("%6d", matrix[i][j]);
        }
        if(i!=rows-1)
        {
            printf("\n");
        }
    }
}

void copy_matrix(int ***matrix, int ***matrix_cop, int *rows, int *cols, int if_print)
{
    *matrix_cop=(int**)malloc(*rows*sizeof(int*));
    for (int i=0; i<*rows; i++)
    {
        (*matrix_cop)[i] = (int*)malloc(*cols*sizeof(int));
    }
    for (int i=0; i <*rows; i++)
    {
        for (int j = 0; j<*cols; j++)
        {
            (*matrix_cop)[i][j]=(*matrix)[i][j];
        }
    }
    if(if_print==1)
    {
        print_matrix(*matrix_cop, *rows, *cols);
    }
}

void create_int_matrix(int **arr, int ***matrix, int *rows, int *cols, int if_print)
{
    scanf("%d %d", rows, cols);
    int count=-1;
    *matrix=(int**)malloc(*rows*sizeof(int*));
    for (int i=0; i<*rows; i++)
    {
        (*matrix)[i]=(int*)malloc(*cols*sizeof(int));
    }
    *arr=malloc(((*rows)*(*cols))*sizeof(int));
    for (int i=0; i<(*rows)*(*cols); i++)
    {
        scanf(" %d", &(*arr)[i]);
    }
    for (int i=0; i<*rows; i++)
    {
        for (int j=0; j<*cols; j++)
        {
            count++;
            (*matrix)[i][j]=(*arr)[count];
        }
    }
    if(if_print==1)
    {
        //printf("\n");
        print_matrix(*matrix, *rows, *cols);
    }
}

void create_int_random_matrix(int ***matrix, int *rows, int *cols, int if_print)
{
    int R, A, B, count=-1;
    scanf("%d %d", rows, cols);
    scanf("%d %d", &A, &B);
    *matrix=(int**)malloc(*rows*sizeof(int*));
    for (int i=0; i<*rows; i++)
    {
        (*matrix)[i]=(int *)malloc(*cols*sizeof(int));
    }
    for (int i=0; i<*rows; i++)
    {
        for (int j=0; j<*cols; j++)
        {
            R=A+rand()%(B-A+1);
            count++;
            (*matrix)[i][j]=R;
        }
    }
    if(if_print==1)
    {
        print_matrix(*matrix, *rows, *cols);
    }
}

void transposition_matrix(int ***matrix, int *rows, int *cols, int if_print)
{
    int flag=-1, t;
    for(int i=0; i<*rows-1; i++)
    {
        flag++;
        for(int j=i+1; j<*cols; j++)
        {
            t=(*matrix)[i][j];
            (*matrix)[i][j]=(*matrix)[j][i];
            (*matrix)[j][i]=t;
        }
    }
    if(if_print==1)
    {
        print_matrix(*matrix, *rows, *cols);
    }
}

void summ_matrix(int ***matrix_1, int ***matrix_2, int ***matrix, int *rows, int *cols, int if_print)
{
    *matrix=(int**)malloc(*rows*sizeof(int*));
    for (int i=0; i<*rows; i++)
    {
        (*matrix)[i]=(int *)malloc(*cols*sizeof(int));
    }
    for (int i=0; i<*rows; i++)
    {
        for (int j=0; j<*cols; j++)
        {
            (*matrix)[i][j]=(*matrix_1)[i][j]+(*matrix_2)[i][j];
        }
    }
    if(if_print==1)
    {
        print_matrix(*matrix, *rows, *cols);
    }
}

void product_matrix(int ***matrix_1, int ***matrix_2, int ***matrix, int *rows_1, int *cols_1, int *rows_2, int *cols_2, int if_print)
{
    int summ;
    if(*cols_1==*rows_2)
    {
        *matrix=(int**)malloc(*rows_1*sizeof(int*));
        for (int i=0; i<*rows_1; i++)
        {
            (*matrix)[i]=(int *)malloc(*cols_2*sizeof(int));
        }


        for(int i=0; i<*rows_1; i++)
        {
            for(int j=0; j<*cols_2; j++)
            {
                summ=0;
                for(int k=0; k<*rows_2; k++)
                {
                    summ=summ+(*matrix_1)[i][k] * (*matrix_2)[k][j];
                }
                (*matrix)[i][j]=summ;
            }
        }
        if(if_print==1)
        {
            print_matrix(*matrix, *rows_1, *cols_2);
        }
    }
}

void raise_matrix_power(int ***matrix, int ***matrix_r, int *rows, int *cols, int power, int if_print) {
    int summ;
    if (*rows == *cols) 
    {
        // Выделение памяти для матрицы результатов
        *matrix_r = (int **)malloc(*rows * sizeof(int *));
        for (int i = 0; i < *rows; i++)
        {
            (*matrix_r)[i] = (int *)malloc(*cols * sizeof(int));
        }
        // Инициализация матрицы результатов как копии исходной
        for (int i = 0; i < *rows; i++)
        {
            for (int j = 0; j < *cols; j++)
            {
                (*matrix_r)[i][j] = (*matrix)[i][j];
            }
        }
        // Выполнение возведения в степень
        for (int f = 1; f < power; f++)
        {
            // Умножение матрицы на саму себя power раз
            int **temp_result = (int **)malloc(*rows * sizeof(int *));
            for (int i = 0; i < *rows; i++)
            {
                temp_result[i] = (int *)malloc(*cols * sizeof(int));
            }

            for (int i = 0; i < *rows; i++)
            {
                for (int j = 0; j < *cols; j++)
                {
                    summ = 0;
                    for (int k = 0; k < *rows; k++)
                    {
                        summ += (*matrix)[i][k] * (*matrix_r)[k][j];
                    }
                    temp_result[i][j] = summ;
                }
            }
            // Обновление матрицы результата
            for (int i = 0; i < *rows; i++)
            {
                for (int j = 0; j < *cols; j++)
                {
                    (*matrix_r)[i][j] = temp_result[i][j];
                }
            }
            // Освобождение памяти временной матрицы
            for (int i = 0; i < *rows; i++)
            {
                free(temp_result[i]);
            }
            free(temp_result);
        }
        if (if_print == 1)
        {
            print_matrix(*matrix_r, *rows, *cols);
        }
        // Обновление оригинальной матрицы, если это необходимо
        if (power>1)
        {
            for (int i = 0; i < *rows; i++)
            {
                for (int j = 0; j < *cols; j++)
                {
                    (*matrix)[i][j] = (*matrix_r)[i][j];
                }
            }
        }
    }
}

    }
	fclose(file);
}
