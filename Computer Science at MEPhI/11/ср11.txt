{
    for(int i=0; i<l; i++)

int f5(int mas[], int l)
{
    int min=mas[0];
    for(int i=0; i<l; i++)
    {
    if (mas[i]<=min) {min=mas[i];}
    }
    return min;
}

            masy[j]+=1;

}
