void f2(int mas[], int l);

{
    int a1, b1, a2, b2, l1=10, l2=12, mas1[l1], mas2[l2];

void f1(int mas[], int l, int a, int b)
{
    for(int i=0; i<l; i++)
    {
        mas[i]=a+b*(i+1);
    }
}

{

    printf("\n");
