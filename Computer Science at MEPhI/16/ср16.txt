#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void read(char str[][20], char *arp[], int *n, int flag)
{
    if(flag==1)
    {
        while (*n < 100 && fgets(str[*n], 20, stdin) && str[*n][0] != '\n')
        {
            str[*n][strcspn(str[*n], "\n")] = '\0'; // Remove newline
            arp[*n] = str[*n];
            (*n)++;
        }
    }
    else if(flag==2)
    {
        FILE *file = fopen("777.txt", "r");
        if (file == NULL)
        {
            perror("Error opening file");
            exit(1);
        }
        while (*n < 100 && fscanf(file, "%19s", str[*n]) == 1)
        {
            arp[*n] = str[*n];
            (*n)++;
        }
        fclose(file);
    }
    else
    {
        printf("Invalid flag selected.\n");
    }
}

void sort_arp(int n, char *arp[n])
{
    char *temp;
    for (int i = 0; i < n-1; i++)
    {
        for (int j = i+1; j < n; j++)
        {
            if (strcmp(arp[i], arp[j]) > 0)
            {
                temp = arp[i];
                arp[i] = arp[j];
                arp[j] = temp;
            }
        }
    }
}

void print_by_arp(int n, char *arp[n])
{
    for(int i = 0; i < n; i++)
    {
        printf("%s\n", arp[i]);
    }
}

void categorize_by_letter(int n, char *arp[n])
{
    int num1 = 0, num2 = 0, num3 = 0;
    for(int i = 0; i < n; i++)
    {
        if(*arp[i] <= 'I') num1++;
        else if (*arp[i] <= 'S') num2++;
        else num3++;
    }
    printf("A-I %d\n", num1);
    printf("J-S %d\n", num2);
    printf("T-Z %d\n", num3);
}

int main()
{
    int n = 0;
    char arr[100][20];
    char *arp[100];
    int flag=1;

    read(arr, arp, &n, flag);
    sort_arp(n, arp);
    print_by_arp(n, arp);
    printf("\n");
    categorize_by_letter(n, arp);

    return 0;
}
